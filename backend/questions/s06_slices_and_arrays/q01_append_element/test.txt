package main
import("sandbox/student";"sandbox/solution";"math/rand";"time";"fmt")
func main() {
	rand.Seed(time.Now().UnixNano())
	for i := 0; i < 8; i++ {
		l := rand.Intn(5)+1
		arr := make([]int, l)
		for j := 0; j < l; j++ { arr[j] = rand.Intn(50) }
		n := rand.Intn(100)
		cp := make([]int, l); copy(cp, arr)
		Compare(solution.AppendElement(arr, n), student.AppendElement(cp, n), fmt.Sprintf("AppendElement(%v, %d)", cp, n))
	}
	Compare(solution.AppendElement([]int{}, 1), student.AppendElement([]int{}, 1), "AppendElement([], 1)")
	Success()
}
