package main
import ("os/exec"; "strings"; "math/rand"; "time"; "fmt")
func main() {
	rand.Seed(time.Now().UnixNano())
	names := []string{"Ali","Sarah","Omar","Lina","Ahmad","Noor","Yusuf","Dana","Sami","Maha"}
	for i := 0; i < 6; i++ {
		name := names[rand.Intn(len(names))]
		out, _ := exec.Command("go", "run", "student/main.go", name).Output()
		Compare(fmt.Sprintf("Hello, %s! Welcome to Go.", name), strings.TrimSpace(string(out)), fmt.Sprintf("Greeting %s", name))
	}
	for i := 0; i < 4; i++ {
		name := GenerateRandomString(rand.Intn(10)+4)
		out, _ := exec.Command("go", "run", "student/main.go", name).Output()
		Compare(fmt.Sprintf("Hello, %s! Welcome to Go.", name), strings.TrimSpace(string(out)), fmt.Sprintf("Greeting %s (random)", name))
	}
	out, _ := exec.Command("go", "run", "student/main.go").Output()
	Compare("Hello, Stranger! Welcome to Go.", strings.TrimSpace(string(out)), "No args -> Stranger")
	Success()
}
