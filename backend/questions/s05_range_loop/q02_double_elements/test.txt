package main
import("sandbox/student";"sandbox/solution";"math/rand";"time";"fmt")
func main() {
	rand.Seed(time.Now().UnixNano())
	for i := 0; i < 8; i++ {
		l := rand.Intn(6)+1
		arr := make([]int, l)
		for j := 0; j < l; j++ { arr[j] = rand.Intn(100)-50 }
		Compare(solution.DoubleElements(arr), student.DoubleElements(arr), fmt.Sprintf("DoubleElements(%v)", arr))
	}
	Compare(solution.DoubleElements([]int{}), student.DoubleElements([]int{}), "DoubleElements([])")
	Compare(solution.DoubleElements([]int{0}), student.DoubleElements([]int{0}), "DoubleElements([0])")
	Success()
}
