package main
import("sandbox/student";"sandbox/solution";"math/rand";"time";"fmt";"strings")
func main() {
	rand.Seed(time.Now().UnixNano())
	pool := []string{"hello","world","go","is","fun","learn","code","build","test","atom","the","great"}
	for i := 0; i < 8; i++ {
		n := rand.Intn(4)+2
		words := make([]string, n)
		for j := 0; j < n; j++ { words[j] = pool[rand.Intn(len(pool))] }
		sentence := strings.Join(words, " ")
		Compare(solution.TitleCase(sentence), student.TitleCase(sentence), fmt.Sprintf("TitleCase(\"%s\")", sentence))
	}
	for i := 0; i < 5; i++ {
		n := rand.Intn(4)+2
		words := make([]string, n)
		for j := 0; j < n; j++ { words[j] = GenerateRandomString(rand.Intn(8)+2) }
		sentence := strings.Join(words, " ")
		Compare(solution.TitleCase(sentence), student.TitleCase(sentence), fmt.Sprintf("TitleCase(\"%s\") (random)", sentence))
	}
	Compare(solution.TitleCase("one"), student.TitleCase("one"), "TitleCase(\"one\")")
	Compare(solution.TitleCase(""), student.TitleCase(""), "TitleCase(\"\")")
	Compare(solution.TitleCase("already Capital"), student.TitleCase("already Capital"), "TitleCase(\"already Capital\")")
	Success()
}
